/************************************************* * CENTRAL SHOP - GOOGLE APPS SCRIPT *************************************************/ const SHEET_NAME = "Orders"; /* * IMPORTANT * ဒီ ၂ ခုကို ကိုယ့်အချက်အလက်နဲ့ ပြောင်းပါ။ */ const BOT_TOKEN = "YOUR_TELEGRAM_BOT_TOKEN"; const ADMIN_CHAT_ID = "YOUR_ADMIN_CHAT_ID"; /************************************************* * GET SHEET *************************************************/ function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow([ "Order ID", "Telegram ID", "Username", "Product", "Price", "Player ID", "Server ID", "Payment", "Status", "Created At", "Updated At" ]); } return sheet; } /************************************************* * DO GET *************************************************/ function doGet(e) { try { const action = e && e.parameter && e.parameter.action ? e.parameter.action : ""; if (action === "getOrders") { return jsonOutput( getOrders( e.parameter.telegramId ) ); } if (action === "getOrder") { return jsonOutput( getOrder( e.parameter.orderId ) ); } return jsonOutput({ success: true, message: "Central Shop API is running." }); } catch (error) { return jsonOutput({ success: false, error: error.toString() }); } } /************************************************* * DO POST *************************************************/ function doPost(e) { try { if (!e || !e.postData) { return jsonOutput({ success: false, error: "No POST data." }); } const data = JSON.parse( e.postData.contents ); const action = data.action || ""; /*********************************************** * CREATE ORDER ***********************************************/ if ( action === "createOrder" ) { return jsonOutput( createOrder(data) ); } /*********************************************** * APPROVE ORDER ***********************************************/ if ( action === "approveOrder" ) { return jsonOutput( updateOrderStatus( data.orderId, "SUCCESS" ) ); } /*********************************************** * CANCEL ORDER ***********************************************/ if ( action === "cancelOrder" ) { return jsonOutput( updateOrderStatus( data.orderId, "FAILED" ) ); } return jsonOutput({ success: false, error: "Unknown action: " + action }); } catch (error) { return jsonOutput({ success: false, error: error.toString() }); } } /************************************************* * CREATE ORDER *************************************************/ function createOrder(data) { const sheet = getSheet(); const orderId = "CS" + Date.now(); const now = new Date(); sheet.appendRow([ orderId, data.telegramId || "", data.username || "", data.product || "", data.price || "", data.playerId || "", data.serverId || "", data.payment || "", "PENDING", now, now ]); /*********************************************** * SEND ORDER TO ADMIN ***********************************************/ sendOrderToAdmin({ orderId: orderId, telegramId: data.telegramId || "", username: data.username || "", product: data.product || "", price: data.price || "", playerId: data.playerId || "", serverId: data.serverId || "", payment: data.payment || "" }); return { success: true, orderId: orderId, status: "PENDING" }; } /************************************************* * UPDATE ORDER STATUS *************************************************/ function updateOrderStatus( orderId, newStatus ) { if (!orderId) { return { success: false, error: "Order ID is missing." }; } const sheet = getSheet(); const values = sheet .getDataRange() .getValues(); for ( let i = 1; i < values.length; i++ ) { const rowOrderId = String(values[i][0]); if ( rowOrderId === String(orderId) ) { /* * Column I = Status */ sheet .getRange(i + 1, 9) .setValue(newStatus); /* * Column K = Updated At */ sheet .getRange(i + 1, 11) .setValue(new Date()); const telegramId = String(values[i][1]); const product = values[i][3]; const price = values[i][4]; /* * Send notification to customer */ sendCustomerStatus( telegramId, orderId, product, price, newStatus ); return { success: true, orderId: orderId, status: newStatus }; } } return { success: false, error: "Order not found: " + orderId }; } /************************************************* * GET CUSTOMER ORDERS *************************************************/ function getOrders( telegramId ) { const sheet = getSheet(); const values = sheet .getDataRange() .getValues(); const orders = []; for ( let i = 1; i < values.length; i++ ) { const row = values[i]; if ( String(row[1]) === String(telegramId) ) { orders.push({ id: String(row[0]), telegramId: String(row[1]), username: String(row[2] || ""), product: String(row[3] || ""), price: String(row[4] || ""), player: String(row[5] || ""), server: String(row[6] || ""), payment: String(row[7] || ""), status: String(row[8] || "PENDING"), time: formatDate(row[9]), updatedAt: formatDate(row[10]) }); } } /* * Newest first */ orders.reverse(); return { success: true, orders: orders }; } /************************************************* * GET SINGLE ORDER *************************************************/ function getOrder( orderId ) { const sheet = getSheet(); const values = sheet .getDataRange() .getValues(); for ( let i = 1; i < values.length; i++ ) { const row = values[i]; if ( String(row[0]) === String(orderId) ) { return { success: true, order: { id: String(row[0]), telegramId: String(row[1]), username: String(row[2] || ""), product: String(row[3] || ""), price: String(row[4] || ""), player: String(row[5] || ""), server: String(row[6] || ""), payment: String(row[7] || ""), status: String(row[8] || "PENDING"), time: formatDate(row[9]), updatedAt: formatDate(row[10]) } }; } } return { success: false, error: "Order not found." }; } /************************************************* * SEND ORDER TO ADMIN *************************************************/ function sendOrderToAdmin( order ) { const text = "🔔 NEW CENTRAL SHOP ORDER\n\n" + "📦 Order ID:\n" + order.orderId + "\n\n" + "👤 Customer:\n" + ( order.username ? "@" + order.username : "Telegram User" ) + "\n\n" + "🆔 Telegram ID:\n" + order.telegramId + "\n\n" + "🎮 Product:\n" + order.product + "\n\n" + "💰 Price:\n" + order.price + "\n\n" + "🎮 Player ID:\n" + order.playerId + "\n\n" + "🌐 Server ID:\n" + ( order.serverId || "-" ) + "\n\n" + "💳 Payment:\n" + order.payment + "\n\n" + "⏳ Status: PENDING"; const keyboard = { inline_keyboard: [ [ { text: "✅ အတည်ပြု", callback_data: "approve:" + order.orderId }, { text: "❌ ပယ်ဖျက်", callback_data: "cancel:" + order.orderId } ] ] }; telegramSendMessage( ADMIN_CHAT_ID, text, keyboard ); } /************************************************* * TELEGRAM CALLBACK *************************************************/ function doTelegramCallback( callback ) { const data = callback.data || ""; const parts = data.split(":"); const action = parts[0]; const orderId = parts.slice(1).join(":"); let result; if ( action === "approve" ) { result = updateOrderStatus( orderId, "SUCCESS" ); } else if ( action === "cancel" ) { result = updateOrderStatus( orderId, "FAILED" ); } else { result = { success: false, error: "Unknown callback." }; } /* * Update admin message */ if ( result.success ) { telegramAnswerCallback( callback.id, action === "approve" ? "✅ Order အောင်မြင်စွာ အတည်ပြုပြီးပါပြီ" : "❌ Order ပယ်ဖျက်ပြီးပါပြီ" ); } else { telegramAnswerCallback( callback.id, "⚠️ " + result.error ); } } /************************************************* * TELEGRAM WEBHOOK *************************************************/ function doPostTelegram(e) { try { const update = JSON.parse( e.postData.contents ); if ( update.callback_query ) { doTelegramCallback( update.callback_query ); } return ContentService .createTextOutput("OK"); } catch (error) { return ContentService .createTextOutput("ERROR"); } } /************************************************* * TELEGRAM SEND MESSAGE *************************************************/ function telegramSendMessage( chatId, text, keyboard ) { if ( !BOT_TOKEN || BOT_TOKEN === "YOUR_TELEGRAM_BOT_TOKEN" ) { console.log( "BOT_TOKEN not configured." ); return; } const url = "https://api.telegram.org/bot" + BOT_TOKEN + "/sendMessage"; const payload = { chat_id: chatId, text: text, parse_mode: "HTML", reply_markup: JSON.stringify( keyboard ) }; UrlFetchApp.fetch( url, { method: "post", contentType: "application/json", payload: JSON.stringify( payload ), muteHttpExceptions: true } ); } /************************************************* * TELEGRAM CALLBACK ANSWER *************************************************/ function telegramAnswerCallback( callbackId, text ) { if ( !BOT_TOKEN || BOT_TOKEN === "YOUR_TELEGRAM_BOT_TOKEN" ) { return; } const url = "https://api.telegram.org/bot" + BOT_TOKEN + "/answerCallbackQuery"; const payload = { callback_query_id: callbackId, text: text, show_alert: false }; UrlFetchApp.fetch( url, { method: "post", contentType: "application/json", payload: JSON.stringify( payload ), muteHttpExceptions: true } ); } /************************************************* * CUSTOMER NOTIFICATION *************************************************/ function sendCustomerStatus( telegramId, orderId, product, price, status ) { if (!telegramId) return; let text = ""; if ( status === "SUCCESS" ) { text = "✅ ORDER SUCCESS\n\n" + "📦 Order ID:\n" + orderId + "\n\n" + "🎮 Product:\n" + product + "\n\n" + "💰 Price:\n" + price + "\n\n" + "🎉 Your order has been successfully completed."; } else if ( status === "FAILED" ) { text = "❌ ORDER FAILED\n\n" + "📦 Order ID:\n" + orderId + "\n\n" + "🎮 Product:\n" + product + "\n\n" + "💰 Price:\n" + price + "\n\n" + "⚠️ Your order was not successful."; } else { text = "⏳ ORDER PENDING\n\n" + "📦 Order ID:\n" + orderId; } telegramSendMessage( telegramId, text, null ); } /************************************************* * FORMAT DATE *************************************************/ function formatDate( value ) { if (!value) return ""; try { return Utilities .formatDate( new Date(value), Session .getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss" ); } catch (e) { return String(value); } } /************************************************* * JSON OUTPUT *************************************************/ function jsonOutput( data ) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); } /************************************************* * WEBHOOK SETUP *************************************************/ function setTelegramWebhook() { const webAppUrl = ScriptApp .getService() .getUrl(); const url = "https://api.telegram.org/bot" + BOT_TOKEN + "/setWebhook?url=" + encodeURIComponent( webAppUrl ); const response = UrlFetchApp.fetch( url ); Logger.log( response.getContentText() ); }
