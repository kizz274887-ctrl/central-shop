const APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbzUJkxWhweDv7RWS3DBrAB8wkS-tplIX6R77w8sIJ6FpknJqH6x4NFyVC8JOZAWjSbBlw/exec"; function submitOrder() { const product = document.getElementById("orderProduct").value; const price = document.getElementById("orderPrice").value; const playerId = document.getElementById("playerId").value.trim(); const serverId = document.getElementById("serverId").value.trim(); const payment = document.getElementById("payment").value; if (!playerId) { tg.showAlert( "⚠️ Please enter Player ID / Username" ); return; } /* Telegram User */ const user = tg.initDataUnsafe && tg.initDataUnsafe.user ? tg.initDataUnsafe.user : null; const telegramId = user ? String(user.id) : ""; const username = user && user.username ? user.username : ""; if (!telegramId) { tg.showAlert( "⚠️ Please open this shop from Telegram." ); return; } /* Prevent double click */ const button = document.querySelector(".confirm"); if (button) { button.disabled = true; button.innerText = "⏳ SENDING ORDER..."; } const orderData = { action: "createOrder", telegramId: telegramId, username: username, product: product, price: price, playerId: playerId, serverId: serverId, payment: payment }; fetch(APPS_SCRIPT_URL, { method: "POST", body: JSON.stringify(orderData) }) .then(response => { return response.json(); }) .then(result => { if (result.success) { tg.showAlert( "✅ ORDER CREATED!\n\n" + "📦 Order ID:\n" + result.orderId + "\n\n" + "🎮 Product:\n" + product + "\n\n" + "💰 Price:\n" + price + "\n\n" + "⏳ Status: PENDING\n\n" + "👤 Admin will confirm your order." ); openPage("ordersPage"); showOrders(); } else { tg.showAlert( "❌ Order failed.\n\n" + (result.error || "Unknown error") ); } }) .catch(error => { console.error(error); tg.showAlert( "❌ Connection error.\n\n" + "Please try again." ); }) .finally(() => { if (button) { button.disabled = false; button.innerText = "🛒 CONFIRM ORDER"; } }); }
